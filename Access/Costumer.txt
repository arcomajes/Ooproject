        private void dataGridView2_CellClick(object sender, DataGridViewCellEventArgs e)
        {

            int rowIndex = e.RowIndex;
            int newQuantity = Convert.ToInt32(dataGridView2.Rows[rowIndex].Cells["Quantity"].Value);
            string productName = dataGridView2.Rows[rowIndex].Cells["Name"].Value.ToString();

            OleDbConnection myConn = new OleDbConnection("Provider=Microsoft.ACE.OLEDB.12.0;Data Source=C:\\Users\\Angel Rose Comajes\\Documents\\Access.accdb");
            myConn.Open();

            string query = "";
            if (productName.StartsWith("Burger"))
            {
                query = "UPDATE Burgers SET Stock = Stock - @1 WHERE Burger = @2";
            }
            else if (productName.StartsWith("Shake"))
            {
                query = "UPDATE Shakes SET Stock = Stock - @1 WHERE Shake = @2";
            }
            else if (productName.StartsWith("AddOn"))
            {
                query = "UPDATE AddOns SET Stock = Stock - @1 WHERE AddOn = @2";
            }

            OleDbCommand cmd = new OleDbCommand(query, myConn);
            cmd.Parameters.AddWithValue("@1", newQuantity);
            cmd.Parameters.AddWithValue("@2", productName);
            cmd.ExecuteNonQuery();

            myConn.Close();

        }

        private void InitializeOrderDataGridView()
        {
            
            dataGridView2.AutoGenerateColumns = false;
            dataGridView2.Columns.Add("Product", "Product");
            dataGridView2.Columns.Add("Price", "Price");
            dataGridView2.Columns.Add("Quantity", "Quantity");
            dataGridView2.Columns.Add("Subtotal", "Subtotal");
            dataGridView2.Columns[0].DataPropertyName = "Product";
            dataGridView2.Columns[1].DataPropertyName = "Price";
            dataGridView2.Columns[2].DataPropertyName = "Quantity";
            dataGridView2.Columns[3].DataPropertyName = "Subtotal";
        }
        private void iconButtonSum_Click(object sender, EventArgs e)
        {
            decimal totalPrice = 0;

            foreach (DataGridViewRow row in dataGridView2.Rows)
            {
                decimal price = Convert.ToDecimal(row.Cells["Price"].Value);
                int quantity = Convert.ToInt32(row.Cells["Quantity"].Value);
                totalPrice += price * quantity;
            }

            textBoxSum.Text = totalPrice.ToString("0.00");

            decimal cash = 0;
            if (decimal.TryParse(textBoxCash.Text, out cash))
            {
                decimal change = cash - totalPrice;
                textBoxChange.Text = change.ToString("0.00");
            }
        }
